    <script>
        const ADMIN_PHONE = "919799995511";
        let currentSelectedSeat = null;
        let seatsData = JSON.parse(localStorage.getItem('eminence_seats')) || {};

        function saveSeats() {
            localStorage.setItem('eminence_seats', JSON.stringify(seatsData));
            renderLayouts();
        }

        function initLayout() {
            renderLayouts();
        }

        function createSeatNode(num, row, col) {
            const seat = document.createElement("div");
            const info = seatsData[num] || { status: 'vacant' };
            
            let seatImage = "desk1.png";
            if (info.status === 'booked' || info.status === 'waiting') {
                seatImage = "desk2.png";
            }

            seat.className = `seat ${info.status}`;
            seat.style.gridRow = row;
            seat.style.gridColumn = col;
            seat.innerHTML = `<img src="${seatImage}" alt="Seat" class="seat-icon"><span>${num}</span>`;
            seat.onclick = () => onSeatClick(num, info);
            return seat;
        }

        function renderLayouts() {
            // --- HALL 1 RENDERING VIA CSS GRID ---
            const h1Container = document.getElementById("hall1GridContainer");
            h1Container.innerHTML = "";

            // 1. Top Row: 12 to 18 (Row 1, Cols 1 to 7)
            for (let i = 12; i <= 18; i++) {
                h1Container.appendChild(createSeatNode(i, 1, i - 11));
            }

            // 2. Left Column: 1 to 11 (Col 1, Rows 2 to 12)
            for (let i = 11; i >= 1; i--) {
                let rowPos = 13 - i;
                h1Container.appendChild(createSeatNode(i, rowPos, 1));
            }

            // 3. Right Vertical Column: 19 to 32 (Col 8, Rows 1 to 14)
            for (let i = 19; i <= 32; i++) {
                let rowPos = i - 18;
                h1Container.appendChild(createSeatNode(i, rowPos, 8));
            }

            // 4. Center Column: 49 to 57 (Col 3, Rows 3 to 11)
            for (let i = 57, r = 3; i >= 49; i--, r++) {
                h1Container.appendChild(createSeatNode(i, r, 3));
            }

            // 5. Inner Double Block (Left: 58, 41-48 on Col 5, Rows 3 onwards)
            const leftInnerSeats = [58, 41, 42, 43, 44, 45, 46, 47, 48];
            leftInnerSeats.forEach((seatNum, index) => {
                let rowPos = 3 + index; 
                h1Container.appendChild(createSeatNode(seatNum, rowPos, 5));
            });

            // 6. Inner Double Block (Right: 59, 40-33 on Col 6, Rows 3 onwards)
            const rightInnerSeats = [59, 40, 39, 38, 37, 36, 35, 34, 33];
            rightInnerSeats.forEach((seatNum, index) => {
                let rowPos = 3 + index; 
                h1Container.appendChild(createSeatNode(seatNum, rowPos, 6));
            });


            // --- HALL 2 RENDERING VIA CSS GRID ---
            const h2Container = document.getElementById("hall2GridContainer");
            h2Container.innerHTML = "";

            // 1. Top Row: 65 to 70 (Row 1, Cols 1 to 6)
            for (let i = 65; i <= 70; i++) {
                h2Container.appendChild(createSeatNode(i, 1, i - 64));
            }

            // 2. Left Column: 64 to 61 (Col 1, Rows 2 to 5)
            const h2LeftSeats = [64, 63, 62, 61];
            h2LeftSeats.forEach((seatNum, index) => {
                let rowPos = 2 + index;
                h2Container.appendChild(createSeatNode(seatNum, rowPos, 1));
            });

            // 3. Right Column: 71 to 74 (Col 6, Rows 2 to 5)
            const h2RightSeats = [71, 72, 73, 74];
            h2RightSeats.forEach((seatNum, index) => {
                let rowPos = 2 + index;
                h2Container.appendChild(createSeatNode(seatNum, rowPos, 6));
            });

            // 4. Center Double Block (Left Sub-column: 80, 79, 78 on Col 3, Rows 3 to 5)
            const h2LeftSub = [80, 79, 78];
            h2LeftSub.forEach((seatNum, index) => {
                let rowPos = 3 + index;
                h2Container.appendChild(createSeatNode(seatNum, rowPos, 3));
            });

            // 5. Center Double Block (Right Sub-column: 77, 76, 75 on Col 4, Rows 3 to 5)
            const h2RightSub = [77, 76, 75];
            h2RightSub.forEach((seatNum, index) => {
                let rowPos = 3 + index;
                h2Container.appendChild(createSeatNode(seatNum, rowPos, 4));
            });

            // 6. Bottom Row: 81, 82, 83 (Row 6)
            h2Container.appendChild(createSeatNode(81, 6, 3));
            h2Container.appendChild(createSeatNode(82, 6, 4));
            h2Container.appendChild(createSeatNode(83, 6, 6));
        }

        function onSeatClick(seatNum, info) {
            if (info.status === 'booked') {
                alert(`Seat #${seatNum} is already Booked.`);
                return;
            }
            if (info.status === 'waiting') {
                alert(`Seat #${seatNum} is currently under verification/waiting.`);
                return;
            }

            currentSelectedSeat = seatNum;
            document.getElementById("selectedSeatNum").innerText = seatNum;
            document.getElementById("studentModal").style.display = "flex";
        }

        function handleStudentSubmit(e) {
            e.preventDefault();
            const name = document.getElementById("sName").value.trim();
            const mobile = document.getElementById("sMobile").value.trim();
            const uid = document.getElementById("sUid").value.trim() || 'N/A';
            const feeStatus = document.getElementById("sFeeStatus").value;

            seatsData[currentSelectedSeat] = {
                status: 'waiting',
                studentName: name,
                mobile: mobile,
                uid: uid,
                feeStatus: feeStatus
            };

            saveSeats();
            closeModal('studentModal');

            const msg = `*NEW SEAT REQUEST*\n\nSeat No: ${currentSelectedSeat}\nName: ${name}\nMobile: ${mobile}\nUID: ${uid}\nFee Status: ${feeStatus}\n\nPlease verify and allot from Admin Panel.`;
            const waLink = `https://wa.me/${ADMIN_PHONE}?text=${encodeURIComponent(msg)}`;
            
            alert(`Seat #${currentSelectedSeat} Verification ke liye bhej di gayi hai! Admin ko notify karne ke liye OK par click karein.`);
            window.open(waLink, '_blank');

            document.getElementById("bookingForm").reset();
        }

        /* ADMIN FUNCTIONS */
        function openAdminModal() {
            document.getElementById("adminModal").style.display = "flex";
        }

        function loginAdmin() {
            const pass = document.getElementById("adminPass").value;
            if (pass === "admin123") {
                document.getElementById("adminLoginSec").style.display = "none";
                document.getElementById("adminControlSec").style.display = "block";
            } else {
                alert("Incorrect Password!");
            }
        }

        function applySeatChange() {
            const seatNo = document.getElementById("manageSeatNo").value;
            const newStatus = document.getElementById("manageSeatStatus").value;

            if (!seatNo || (seatNo < 1 || (seatNo > 59 && seatNo < 61) || seatNo > 83)) {
                alert("Please enter a valid Seat Number (1-59 or 61-83)");
                return;
            }

            if (!seatsData[seatNo]) seatsData[seatNo] = {};
            seatsData[seatNo].status = newStatus;

            saveSeats();
            alert(`Seat #${seatNo} Updated! Status: ${newStatus.toUpperCase()}`);

            if (newStatus === 'booked' && seatsData[seatNo].mobile) {
                document.getElementById("btnNotifyStudent").style.display = "block";
            } else {
                document.getElementById("btnNotifyStudent").style.display = "none";
            }
        }

        function notifyStudentWhatsApp() {
            const seatNo = document.getElementById("manageSeatNo").value;
            const student = seatsData[seatNo];

            if (student && student.mobile) {
                const msg = `*Eminence Library - Seat Allotment Success*\n\nHello ${student.studentName},\nYour Seat #${seatNo} has been successfully verified and ALLOTTED to you.\n\nThank You!`;
                window.open(`https://wa.me/91${student.mobile}?text=${encodeURIComponent(msg)}`, '_blank');
            } else {
                alert("No student contact details found for this seat.");
            }
        }

        function searchMember() {
            const query = document.getElementById("searchMemberInput").value.trim().toLowerCase();
            const res = document.getElementById("adminSearchResult");
            let found = [];

            for (let seat in seatsData) {
                let s = seatsData[seat];
                if (s.mobile === query || (s.uid && s.uid.toLowerCase() === query)) {
                    found.push(`Seat #${seat}: ${s.studentName} (${s.status.toUpperCase()})`);
                }
            }

            if (found.length > 0) {
                res.style.color = "green";
                res.innerHTML = found.join("<br>");
            } else {
                res.style.color = "red";
                res.innerHTML = "No record found for this Mobile/UID.";
            }
        }

        function resetAllSeats() {
            if (confirm("Kya aap sach me poora Seat System reset karna chahte hain? Sabhi seats VACANT ho jayengi.")) {
                seatsData = {};
                saveSeats();
                alert("System Reset Successfully!");
            }
        }

        function closeModal(id) {
            document.getElementById(id).style.display = "none";
        }

        window.onload = initLayout;
    </script>
